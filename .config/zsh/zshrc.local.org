# Claude Worktree
_cw_set_pane_title() {
  local type=$1 num=$2
  [[ -z "$TMUX" ]] && return 0
  local title
  if [[ "$type" == "issue" ]]; then
    title=$(gh -R KazuakiM/xxxxxxxxx issue view "$num" --json title -q .title 2>/dev/null)
  else
    title=$(gh -R KazuakiM/xxxxxxxxx pr view "$num" --json title -q .title 2>/dev/null)
  fi
  if [[ -n "$title" ]]; then
    local label="[${type}#${num}] ${title}"
    # Claude 起動で pane_title が上書きされるため、独自 user option に保持して
    # tmux.conf の pane-border-format から優先表示する
    tmux set-option -p @cw_title "$label"
    tmux select-pane -T "$label"
  fi
}

appdir=""
_cw_devcontainer_label() {
  local type=${1:?"Usage: _cw_devcontainer_label <issue|pr> <number>"}
  local num=${2:?"Usage: _cw_devcontainer_label <issue|pr> <number>"}
  echo "${type}-${num}"
}

_cw_require_devcontainer() {
  if ! command -v docker >/dev/null 2>&1; then
    echo "cw: docker が見つかりません。Docker Desktop などを起動し、docker CLI を PATH に追加してください。"
    return 1
  fi

  if ! command -v node >/dev/null 2>&1; then
    echo "cw: Node.js が見つかりません。Node.js 18 以上を有効にしてから devcontainer CLI を使ってください。"
    return 1
  fi

  if ! node -e 'require("crypto").randomUUID()' >/dev/null 2>&1; then
    echo "cw: 現在の Node.js は devcontainer CLI には古すぎます: $(node -v 2>/dev/null)"
    echo "    Node.js 18 以上を有効にしてから @devcontainers/cli を入れ直してください:"
    echo "    nvm install 20 && nvm use 20 && npm install -g @devcontainers/cli"
    return 1
  fi

  if ! command -v devcontainer >/dev/null 2>&1; then
    echo "cw: devcontainer CLI が見つかりません。Node.js を用意してから次を実行してください:"
    echo "    npm install -g @devcontainers/cli"
    return 1
  fi
}

_cw_codex() {
  local wtdir=${1:?"Usage: _cw_codex <worktree-dir> <label>"}
  local label=${2:?"Usage: _cw_codex <worktree-dir> <label>"}
  local config="$appdir/.devcontainer/worktree/devcontainer.json"

  if [[ ! -d "$wtdir" ]]; then
    echo "cw: worktree が見つかりません: $wtdir"
    return 1
  fi

  if [[ ! -f "$config" ]]; then
    echo "cw: Dev Container 設定が見つかりません: $config"
    return 1
  fi

  _cw_require_devcontainer || return 1

  MARKET_JOURNAL_APPDIR="$appdir" devcontainer up \
    --workspace-folder "$wtdir" \
    --config "$config" \
    --id-label "market-journal.repo=market-journal" \
    --id-label "market-journal.worktree=$label" || return 1

  MARKET_JOURNAL_APPDIR="$appdir" devcontainer exec \
    --workspace-folder "$wtdir" \
    --config "$config" \
    --id-label "market-journal.repo=market-journal" \
    --id-label "market-journal.worktree=$label" \
    /bin/bash -lc "cd '$wtdir' && exec codex --cd '$wtdir' --dangerously-bypass-approvals-and-sandbox"
}

_cw_remove_devcontainer() {
  local label=${1:?"Usage: _cw_remove_devcontainer <label>"}
  local ids output

  command -v docker >/dev/null 2>&1 || return 0

  output=$(docker ps -aq \
    --filter "label=market-journal.repo=market-journal" \
    --filter "label=market-journal.worktree=$label" 2>/dev/null) || {
    echo "cw: docker に接続できないため devcontainer 削除をスキップします: $label"
    return 0
  }

  [[ -n "$output" ]] || return 0
  ids=("${(@f)output}")

  if (( ${#ids[@]} > 0 )); then
    docker rm -f "${ids[@]}" >/dev/null || return 1
  fi
}


cw() {
  local num=${1:?"Usage: cw <issue-number>"}
  local branch="feature/$num"
  local wtdir="$appdir/.worktrees/$num"
  local label="$(_cw_devcontainer_label issue "$num")"

  _cw_require_devcontainer || return 1

  git -C "$appdir" fetch origin && \
  git -C "$appdir" worktree add "$wtdir" -b "$branch" origin/main && \
  _cw_set_pane_title issue "$num"
  cd "$wtdir" && _cw_codex "$wtdir" "$label"
}

cwr() {
  local num=${1:?"Usage: cwr <issue-number>"}
  local wtdir="$appdir/.worktrees/$num"

  if [[ ! -d "$wtdir" ]]; then
    echo "cwr: worktree $num not found. Run: cw $num"
    return 1
  fi

  _cw_set_pane_title issue "$num"
  cd "$wtdir" && _cw_codex "$wtdir" "$(_cw_devcontainer_label issue "$num")"
}

cwl() {
  git -C "$appdir" worktree list
}

cwd() {
  local arg=${1:?"Usage: cwd <issue-or-pr-number> | cwd pr-<pr-number>"}
  local wtdir branch label force_flag

  if [[ "$arg" == pr-* ]]; then
    wtdir="$appdir/.worktrees/$arg"
    branch="$arg"
    label="$arg"
    force_flag="-D"
  else
    local wtdir_issue="$appdir/.worktrees/$arg"
    local wtdir_pr="$appdir/.worktrees/pr-$arg"

    if [[ -d "$wtdir_issue" && -d "$wtdir_pr" ]]; then
      echo "cwd: $arg と pr-$arg の worktree が両方存在します。明示してください: cwd pr-$arg"
      return 1
    elif [[ -d "$wtdir_pr" ]]; then
      wtdir="$wtdir_pr"; branch="pr-$arg"; label="pr-$arg"; force_flag="-D"
    elif [[ -d "$wtdir_issue" ]]; then
      wtdir="$wtdir_issue"; branch="feature/$arg"; label="$arg"; force_flag="-d"
    else
      echo "cwd: worktree が見つかりません ($arg / pr-$arg のどちらにも存在しない)"
      return 1
    fi
  fi

  if [[ ! -d "$wtdir" ]]; then
    echo "cwd: worktree が見つかりません: $wtdir"
    return 1
  fi

  if [[ "$label" == pr-* ]]; then
    _cw_remove_devcontainer "$(_cw_devcontainer_label pr "${label#pr-}")" || return 1
  else
    _cw_remove_devcontainer "$(_cw_devcontainer_label issue "$label")" || return 1
  fi

  git -C "$appdir" worktree remove --force "$wtdir" || return 1

  if git -C "$appdir" show-ref --verify --quiet "refs/heads/$branch"; then
    if ! git -C "$appdir" branch "$force_flag" "$branch"; then
      echo "cwd: ブランチ $branch を削除できませんでした (未マージ作業がある可能性)。"
      echo "      強制削除する場合: git -C \"$appdir\" branch -D $branch"
      return 1
    fi
  fi

  echo "cwd: worktree and branch for $label removed."
}

cwp() {
  local num=${1:?"Usage: cwp <pr-number>"}
  local branch="pr-$num"
  local wtdir="$appdir/.worktrees/pr-$num"

  _cw_require_devcontainer || return 1

  if [[ -d "$wtdir" ]]; then
    git -C "$wtdir" fetch origin && \
    git -C "$wtdir" fetch -f origin "pull/$num/head" && \
    git -C "$wtdir" reset --hard FETCH_HEAD || return 1
  else
    git -C "$appdir" fetch origin && \
    git -C "$appdir" fetch -f origin "pull/$num/head:$branch" && \
    git -C "$appdir" worktree add "$wtdir" "$branch" || return 1
  fi

  git -C "$wtdir" rebase origin/main || {
    echo "cwp: rebase でコンフリクトが発生しました。$wtdir で解決してください。"
    cd "$wtdir"
    return 1
  }

  _cw_set_pane_title pr "$num"
  cd "$wtdir" && _cw_codex "$wtdir" "$(_cw_devcontainer_label pr "$num")"
}

cwj() {
  local arg=${1:?"Usage: cwj <issue-or-pr-number> | cwj pr-<pr-number>"}
  local wtdir type num

  if [[ "$arg" == pr-* ]]; then
    wtdir="$appdir/.worktrees/$arg"
    type="pr"
    num="${arg#pr-}"
  else
    local wtdir_issue="$appdir/.worktrees/$arg"
    local wtdir_pr="$appdir/.worktrees/pr-$arg"

    if [[ -d "$wtdir_issue" && -d "$wtdir_pr" ]]; then
      echo "cwj: $arg と pr-$arg の worktree が両方存在します。明示してください: cwj pr-$arg"
      return 1
    elif [[ -d "$wtdir_pr" ]]; then
      wtdir="$wtdir_pr"; type="pr"; num="$arg"
    elif [[ -d "$wtdir_issue" ]]; then
      wtdir="$wtdir_issue"; type="issue"; num="$arg"
    else
      echo "cwj: worktree が見つかりません ($arg / pr-$arg のどちらにも存在しない)"
      return 1
    fi
  fi

  _cw_set_pane_title "$type" "$num"
  cd "$wtdir"
}
